CREATE DATABASE shopping_management;

USE shopping_management;

CREATE TABLE products (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL,
    price DECIMAL(10,2) NOT NULL CHECK (price > 0),
    stock_quantity INT NOT NULL CHECK (stock_quantity >= 0),
    supplier VARCHAR(100),
    created_date DATE
);

CREATE TABLE customers (
    customer_id INT PRIMARY KEY,
    customer_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    city VARCHAR(50),
    age INT CHECK (age >= 18),
    registration_date DATE
);

CREATE TABLE sales (
    sale_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT NOT NULL CHECK (quantity > 0),
    total_amount DECIMAL(10,2) NOT NULL,
    payment_mode VARCHAR(30) DEFAULT 'Cash',
    sale_date DATE,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id),
    FOREIGN KEY (product_id) REFERENCES products(product_id)
);

SHOW DATABASES;

USE shopping_management;

SHOW TABLES;

DESC products;
DESC customers;
DESC sales;

-- MODULE 8: SALES CALCULATIONS

SELECT COUNT(*) AS total_products
FROM products;

SELECT COUNT(*) AS total_customers
FROM customers;

SELECT COUNT(*) AS total_sales
FROM sales;

SELECT SUM(total_amount) AS total_revenue
FROM sales;

SELECT AVG(price) AS average_product_price
FROM products;

SELECT MAX(price) AS highest_product_price
FROM products;

SELECT MIN(price) AS lowest_product_price
FROM products;

SELECT MAX(total_amount) AS highest_sales_transaction
FROM sales;

SELECT SUM(quantity) AS total_quantity_sold
FROM sales;

SELECT AVG(total_amount) AS average_sales_value
FROM sales;


-- MODULE 9: CATEGORY-WISE SALES REPORTS

SELECT category, COUNT(*) AS total_products
FROM products
GROUP BY category;

SELECT category, AVG(price) AS average_price
FROM products
GROUP BY category;

SELECT category, SUM(stock_quantity) AS total_stock
FROM products
GROUP BY category;

SELECT p.category, SUM(s.total_amount) AS total_revenue
FROM products p
INNER JOIN sales s
ON p.product_id = s.product_id
GROUP BY p.category;

SELECT payment_mode, COUNT(*) AS total_transactions
FROM sales
GROUP BY payment_mode;

SELECT p.product_name, SUM(s.quantity) AS total_quantity_sold
FROM products p
INNER JOIN sales s
ON p.product_id = s.product_id
GROUP BY p.product_id, p.product_name;

SELECT category, COUNT(*) AS total_products
FROM products
GROUP BY category
HAVING COUNT(*) > 3;

SELECT p.product_name, SUM(s.quantity) AS total_sold
FROM products p
INNER JOIN sales s
ON p.product_id = s.product_id
GROUP BY p.product_id, p.product_name
HAVING SUM(s.quantity) > 5;

SELECT payment_mode, COUNT(*) AS total_transactions
FROM sales
GROUP BY payment_mode
HAVING COUNT(*) > 5;

SELECT
    DATE_FORMAT(sale_date, '%Y-%m') AS sales_month,
    COUNT(*) AS total_transactions,
    SUM(quantity) AS total_quantity,
    SUM(total_amount) AS monthly_revenue,
    AVG(total_amount) AS average_transaction
FROM sales
GROUP BY DATE_FORMAT(sale_date, '%Y-%m')
ORDER BY sales_month;
